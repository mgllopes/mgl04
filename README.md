# 1. Verificar o status dos arquivos modificados/novos
git status

# 2. Adicionar os artefatos à área de preparação (Staging Area)
git add .

git commit -m "Implementação de novas regras de negócio e ajustes no arquivo novo.py"

# 4. Enviar os commits locais para o repositório remoto
git push origin main
